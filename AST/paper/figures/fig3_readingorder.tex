% #1..#4 = x1,y1,x2,y2 (bottom-left / top-right), #5 = reading position
\newcommand{\region}[5]{%
  \draw[lbox] (#1,#2) rectangle (#3,#4);
  \node[circle, draw=pdfLine!60, fill=white, text=pdfLine, line width=0.5pt,
        inner sep=1.0pt, font=\scriptsize] at ({#1+0.28},{#4-0.28}) {#5};}
